* ---------------------------------------------------------------------
* COREBANK CICS RESOURCE DEFINITIONS - INPUT FOR DFHCSDUP
* (UNTESTED HERE - NO CICS IN THIS ENVIRONMENT)
* ---------------------------------------------------------------------
DEFINE GROUP(COREBANK) DESCRIPTION(COREBANK ONLINE BANKING)

DEFINE TRANSACTION(CBSN) GROUP(COREBANK) PROGRAM(CBSIGN)
       DESCRIPTION(SIGN ON)
DEFINE TRANSACTION(CBMN) GROUP(COREBANK) PROGRAM(CBMENU)
       DESCRIPTION(MAIN MENU)
DEFINE TRANSACTION(CBAI) GROUP(COREBANK) PROGRAM(CBINQ)
       DESCRIPTION(ACCOUNT INQUIRY)
DEFINE TRANSACTION(CBFT) GROUP(COREBANK) PROGRAM(CBXFR)
       DESCRIPTION(FUND TRANSFER)

DEFINE PROGRAM(CBSIGN) GROUP(COREBANK) LANGUAGE(COBOL)
DEFINE PROGRAM(CBMENU) GROUP(COREBANK) LANGUAGE(COBOL)
DEFINE PROGRAM(CBINQ)  GROUP(COREBANK) LANGUAGE(COBOL)
DEFINE PROGRAM(CBXFR)  GROUP(COREBANK) LANGUAGE(COBOL)

DEFINE MAPSET(CBSNMS) GROUP(COREBANK)
DEFINE MAPSET(CBMNMS) GROUP(COREBANK)
DEFINE MAPSET(CBAIMS) GROUP(COREBANK)
DEFINE MAPSET(CBFTMS) GROUP(COREBANK)

* ACCOUNT MASTER: THE SAME KSDS THE BATCH USES (SEE jcl/CBSETUP.jcl).
* RECOVERABLE SO A ROLLBACK UNDOES BOTH SIDES OF A TRANSFER.
DEFINE FILE(ACCTMAST) GROUP(COREBANK)
       DSNAME(CBANK.PROD.ACCTMAST.KSDS)
       ADD(NO) BROWSE(YES) DELETE(NO) READ(YES) UPDATE(YES)
       RECOVERY(BACKOUTONLY) STRINGS(5)

* TRANSACTION LOG AS AN ESDS, WITH A PATH OVER AN ALTERNATE INDEX ON
* THE ACCOUNT NUMBER FOR THE INQUIRY SCREEN (SEE jcl/CICSFILE.jcl)
DEFINE FILE(TRANLOG) GROUP(COREBANK)
       DSNAME(CBANK.PROD.TRANLOG.ESDS)
       ADD(YES) BROWSE(NO) READ(NO) UPDATE(NO)
       RECOVERY(BACKOUTONLY) STRINGS(5)
DEFINE FILE(TRANACCT) GROUP(COREBANK)
       DSNAME(CBANK.PROD.TRANLOG.PATH)
       ADD(NO) BROWSE(YES) READ(YES) UPDATE(NO) STRINGS(5)

ADD GROUP(COREBANK) LIST(CBLIST)
